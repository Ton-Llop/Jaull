"""Offline diagnostic: production decode loop versus greedy model.generate."""

from __future__ import annotations

import argparse
import hashlib
import json
import os
import platform
import statistics
import subprocess
import threading
import time
from collections import Counter
from datetime import UTC, datetime
from pathlib import Path


def emit(path, event, **values):
    row = {"at": datetime.now(UTC).isoformat(), "event": event, **values}
    with path.open("a", encoding="utf-8") as handle:
        handle.write(json.dumps(row, allow_nan=False) + "\n")
    print(json.dumps(row, allow_nan=False), flush=True)


def gpu_state():
    fields = "temperature.gpu,utilization.gpu,memory.used,clocks.sm,clocks.mem,power.draw"
    result = subprocess.run(
        ["nvidia-smi", f"--query-gpu={fields}", "--format=csv,noheader,nounits"],
        capture_output=True, text=True, timeout=10, check=False,
    )
    return {"fields": fields.split(","), "rows": result.stdout.strip().splitlines(),
            "exit_code": result.returncode}


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--rounds", type=int, default=3)
    args = parser.parse_args()
    if args.rounds < 1:
        parser.error("rounds must be positive")
    args.output.mkdir(parents=True, exist_ok=False)
    events = args.output / "events.jsonl"
    emit(events, "start", gpu_before=gpu_state())

    import psutil
    import torch
    import transformers
    from huggingface_hub import snapshot_download

    from jaull.runtime.transformers_benchmark_worker import _decode_once

    if not torch.cuda.is_available():
        raise RuntimeError("CUDA is unavailable to this process")
    snapshot = Path(snapshot_download(
        "Qwen/Qwen2.5-0.5B-Instruct", revision="main", local_files_only=True,
    ))
    manifest = {}
    for path in sorted(snapshot.iterdir()):
        if path.is_file() and (path.suffix in {".safetensors", ".json"}):
            digest = hashlib.sha256()
            with path.open("rb") as handle:
                for chunk in iter(lambda: handle.read(1 << 20), b""):
                    digest.update(chunk)
            manifest[path.name] = {"bytes": path.stat().st_size, "sha256": digest.hexdigest()}
    metadata = {
        "python": platform.python_version(), "platform": platform.platform(),
        "torch": torch.__version__, "transformers": transformers.__version__,
        "cuda_runtime": torch.version.cuda, "gpu": torch.cuda.get_device_name(0),
        "torch_threads": torch.get_num_threads(), "torch_interop": torch.get_num_interop_threads(),
        "affinity": sorted(os.sched_getaffinity(0)), "snapshot": str(snapshot),
        "manifest": manifest,
        "comparison_scope": (
            "One model instance, FP16 CUDA, greedy decoding, same raw synthetic prompt. "
            "Production _decode_once times 128 cached forward steps after prefill and first "
            "token selection. generate produces 129 new tokens: first from prefill plus "
            "128 cached steps. Its steady timer begins at the second forward call; its "
            "end-to-end timer includes prefill. No chat template or sampling."
        ),
        "gpu_observation_scope": "NVML whole-device telemetry, not process-attributed memory",
    }
    with (args.output / "metadata.json").open("x", encoding="utf-8") as handle:
        json.dump(metadata, handle, indent=2, allow_nan=False)
    tokenizer = transformers.AutoTokenizer.from_pretrained(snapshot, local_files_only=True)
    started = time.perf_counter()
    model = transformers.AutoModelForCausalLM.from_pretrained(
        snapshot, local_files_only=True, torch_dtype=torch.float16,
    )
    model.to("cuda")
    torch.cuda.synchronize()
    placement = {
        "parameter_devices": dict(Counter(str(p.device) for p in model.parameters())),
        "parameter_dtypes": dict(Counter(str(p.dtype) for p in model.parameters())),
        "parameter_bytes": sum(p.numel() * p.element_size() for p in model.parameters()),
        "buffer_devices": dict(Counter(str(b.device) for b in model.buffers())),
        "training": model.training,
        "attention": getattr(model.config, "_attn_implementation", None),
        "hf_device_map": getattr(model, "hf_device_map", None),
        "load_seconds": time.perf_counter() - started,
    }
    emit(events, "placement", **placement)
    if set(placement["parameter_devices"]) != {"cuda:0"}:
        raise RuntimeError("Not all parameters reside on CUDA0")
    if set(placement["parameter_dtypes"]) != {"torch.float16"}:
        raise RuntimeError("The model is not wholly FP16")
    if model.training:
        raise RuntimeError("The loaded model is in training mode")

    inputs = tokenizer("Benchmark generation.", return_tensors="pt")
    inputs = {key: value.to("cuda") for key, value in inputs.items()}
    generation = transformers.GenerationConfig(
        do_sample=False, num_beams=1, use_cache=True, max_new_tokens=129,
        min_new_tokens=129, eos_token_id=None, forced_eos_token_id=None,
        pad_token_id=tokenizer.eos_token_id,
        repetition_penalty=1.0, encoder_repetition_penalty=1.0, no_repeat_ngram_size=0,
        temperature=1.0, top_p=1.0, top_k=50,
    )
    effective, _ = model._prepare_generation_config(generation)
    emit(events, "effective_generation", repetition_penalty=effective.repetition_penalty,
         do_sample=effective.do_sample, num_beams=effective.num_beams,
         min_new_tokens=effective.min_new_tokens, max_new_tokens=effective.max_new_tokens,
         temperature=effective.temperature, top_p=effective.top_p, top_k=effective.top_k)

    def runner():
        seconds, tokens = _decode_once(
            torch=torch, tokenizer=tokenizer, model=model,
            prompt="Benchmark generation.", input_device=torch.device("cuda:0"),
            max_new_tokens=128,
        )
        return {"steady_seconds": seconds, "steady_tokens": tokens,
                "steady_tps": tokens / seconds}

    def generate():
        forward_calls = 0
        steady_start = None

        def prehook(module, positional, keywords):
            nonlocal forward_calls, steady_start
            forward_calls += 1
            if forward_calls == 2:
                torch.cuda.synchronize()
                steady_start = time.perf_counter()

        hook = model.register_forward_pre_hook(prehook, with_kwargs=True)
        try:
            torch.cuda.synchronize()
            start = time.perf_counter()
            with torch.inference_mode():
                output = model.generate(**inputs, generation_config=generation)
            torch.cuda.synchronize()
            end = time.perf_counter()
        finally:
            hook.remove()
        tokens = int(output.shape[-1] - inputs["input_ids"].shape[-1])
        if tokens != 129 or forward_calls != 129 or steady_start is None:
            raise RuntimeError(f"Unexpected generate workload: {tokens=}, {forward_calls=}")
        seconds = end - steady_start
        return {"steady_seconds": seconds, "steady_tokens": 128,
                "steady_tps": 128 / seconds, "total_seconds": end - start,
                "total_tokens": tokens, "total_tps": tokens / (end - start),
                "forward_calls": forward_calls}

    for name, call in (("runner", runner), ("generate", generate)):
        emit(events, "warmup", method=name, result=call())

    results = []
    process = psutil.Process()
    for number in range(args.rounds):
        order = (("runner", runner), ("generate", generate))
        if number % 2:
            order = tuple(reversed(order))
        for name, call in order:
            time.sleep(2)
            before_gpu = gpu_state()
            samples = []
            stop = threading.Event()

            def monitor():
                while not stop.wait(1):
                    samples.append({"elapsed": time.perf_counter() - wall_start,
                                    "gpu": gpu_state()})

            cpu_start = sum(process.cpu_times()[:2])
            wall_start = time.perf_counter()
            thread = threading.Thread(target=monitor, daemon=True)
            thread.start()
            try:
                result = call()
            finally:
                stop.set()
                thread.join(timeout=15)
            result.update({"method": name, "round": number + 1,
                           "wall_seconds": time.perf_counter() - wall_start,
                           "cpu_seconds": sum(process.cpu_times()[:2]) - cpu_start,
                           "torch_allocated_bytes": torch.cuda.memory_allocated(),
                           "torch_reserved_bytes": torch.cuda.memory_reserved(),
                           "gpu_before": before_gpu, "gpu_during": samples,
                           "gpu_after": gpu_state()})
            results.append(result)
            emit(events, "measurement", **result)

    traces = {}
    for name, call in (("runner", runner), ("generate", generate)):
        ids = []

        def capture(module, positional, keywords):
            ids.append(keywords["input_ids"].detach().clone())

        hook = model.register_forward_pre_hook(capture, with_kwargs=True)
        try:
            call()
        finally:
            hook.remove()
        traces[name] = [row.cpu().tolist() for row in ids]
    path_equal = traces["runner"] == traces["generate"]
    mismatches = [index for index, (left, right) in enumerate(zip(
        traces["runner"], traces["generate"], strict=True,
    )) if left != right]
    emit(events, "token_path_check", equal=path_equal,
         calls={name: len(trace) for name, trace in traces.items()},
         first_mismatch=mismatches[0] if mismatches else None)
    summary = {"placement": placement, "results": results,
               "consumed_token_paths_equal": path_equal,
               "medians": {name: statistics.median(row["steady_tps"] for row in results
                                                   if row["method"] == name)
                           for name in ("runner", "generate")}}
    with (args.output / "summary.json").open("x", encoding="utf-8") as handle:
        json.dump(summary, handle, indent=2, allow_nan=False)
    emit(events, "finished", medians=summary["medians"], token_paths_equal=path_equal)


if __name__ == "__main__":
    main()
